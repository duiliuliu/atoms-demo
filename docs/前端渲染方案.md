# 前端渲染方案

## 综述

### 功能列表

| 功能模块 | 功能点 | 优先级 |
|---------|--------|--------|
| 实时渲染 | iframe沙箱、热更新、渲染同步 | P0 |
| 一键部署 | 构建打包、CDN上传、域名配置、部署状态 | P0 |
| 页面选择 | 可视化选取、DOM操作、问题标注 | P0 |
| Console展示 | 实时日志、错误捕获、性能监控 | P1 |
| 代码/渲染切换 | 模式切换动画、状态保持、快捷键 | P0 |
| 版本切换 | 版本树、版本对比、快速切换 | P1 |

### 优先级说明
- **P0**：核心渲染能力，用户开发体验的基础
- **P1**：增强功能，提升开发效率

#### 1.3 热更新机制

```typescript
class HotReloadService {
  private ws: WebSocket;
  private hmrClient: HMRClient;
  
  async connect(sandboxId: string): Promise<void> {
    // 1. 建立WebSocket连接
    this.ws = new WebSocket(`wss://sandbox/${sandboxId}/hmr`);
    
    // 2. 初始化HMR客户端
    this.hmrClient = new HMRClient({
      onConnect: () => this.onConnect(),
      onUpdate: (update) => this.applyUpdate(update),
      onError: (error) => this.handleError(error),
    });
    
    // 3. 监听文件变化
    this.ws.on('message', (data) => this.hmrClient.process(data));
  }
  
  async applyUpdate(update: HMRUpdate): Promise<void> {
    switch (update.type) {
      case 'js':
        await this.applyJSUpdate(update);
        break;
      case 'css':
        await this.applyCSSUpdate(update);
        break;
      case 'reload':
        await this.reloadPage();
        break;
      case 'custom':
        await this.applyCustomUpdate(update);
        break;
    }
  }
  
  private async applyJSUpdate(update: HMRUpdate): Promise<void> {
    // 动态导入新模块
    const module = await import(/* @vite-ignore */ update.url);
    
    // HMR API处理
    if (module && module.__hmr) {
      module.__hmr();
    }
    
    // 通知iframe更新完成
    this.postMessage({
      type: 'hmr:updated',
      module: update.module,
      timestamp: Date.now(),
    });
  }
  
  private async applyCSSUpdate(update: HMRUpdate): Promise<void> {
    // 查找对应的style/link标签
    const existing = document.querySelector(
      `link[href*="${update.module}"]`
    );
    
    if (existing) {
      // 更新时间戳，触发重新加载
      existing.href = update.url + `?t=${Date.now()}`;
    }
  }
}

// iframe通信协议
interface IframeProtocol {
  // 发送到iframe的消息
  toIframe: {
    'refresh': void;
    'update:css': { path: string; content: string };
    'update:js': { path: string; url: string };
    'inspect:enable': void;
    'inspect:disable': void;
    'device:change': DeviceConfig;
  };
  
  // 从iframe接收的消息
  fromIframe: {
    'ready': void;
    'error': { message: string; stack: string };
    'console': ConsoleMessage;
    'inspect:select': { selector: string; rect: DOMRect };
    'navigation': { url: string };
  };
}
```

#### 2.2 部署服务实现

```typescript
class DeployService {
  private buildService: BuildService;
  private cdnService: CDNService;
  private domainService: DomainService;
  
  async deploy(request: DeployRequest): Promise<DeployResult> {
    const { projectId, customDomain, userId } = request;
    
    try {
      // 1. 预检查
      const checkResult = await this.preCheck(userId, projectId);
      if (!checkResult.success) {
        return { success: false, error: checkResult.error };
      }
      
      // 2. 构建
      this.emit('status', { phase: 'building', progress: 0 });
      const buildResult = await this.buildService.build(projectId);
      this.emit('status', { phase: 'building', progress: 100 });
      
      // 3. 验证
      this.emit('status', { phase: 'validating', progress: 0 });
      await this.validate(buildResult);
      this.emit('status', { phase: 'validating', progress: 100 });
      
      // 4. 上传
      this.emit('status', { phase: 'uploading', progress: 0 });
      const cdnResult = await this.uploadAssets(buildResult.assets);
      this.emit('status', { phase: 'uploading', progress: 100 });
      
      // 5. 部署
      this.emit('status', { phase: 'deploying', progress: 0 });
      const deployResult = await this.deployToEdge(cdnResult);
      this.emit('status', { phase: 'deploying', progress: 100 });
      
      // 6. 配置域名
      let url = deployResult.url;
      if (customDomain) {
        url = await this.configureCustomDomain(customDomain, cdnResult);
      }
      
      return {
        success: true,
        url,
        deploymentId: deployResult.id,
        expiresAt: deployResult.expiresAt,
      };
      
    } catch (error) {
      return {
        success: false,
        error: this.formatError(error),
      };
    }
  }
  
  private async uploadAssets(assets: BuildAsset[]): Promise<CDNResult> {
    const results = await Promise.all(
      assets.map(async (asset, index) => {
        this.emit('status', {
          phase: 'uploading',
          progress: (index / assets.length) * 100,
          currentFile: asset.path,
        });
        
        return this.cdnService.upload(asset);
      })
    );
    
    return {
      baseUrl: results[0].baseUrl,
      assets: results,
    };
  }
  
  private async deployToEdge(cdnResult: CDNResult): Promise<EdgeDeployResult> {
    // 边缘函数部署（如有后端）
    const edgeFunctions = await this.detectEdgeFunctions();
    
    if (edgeFunctions.length > 0) {
      await Promise.all(
        edgeFunctions.map(fn => this.deployEdgeFunction(fn))
      );
    }
    
    return {
      id: generateId(),
      url: this.generateUrl(),
      expiresAt: Date.now() + 365 * 24 * 60 * 60 * 1000, // 1年
    };
  }
}
```

#### 3.2 选取实现机制

```typescript
class InspectService {
  private iframe: HTMLIFrameElement;
  private overlay: HTMLElement;
  private selectedElement: Element | null;
  
  enable(): void {
    // 1. 创建覆盖层
    this.createOverlay();
    
    // 2. 注入检测脚本
    this.injectInspectScript();
    
    // 3. 监听鼠标移动
    this.iframe.contentWindow?.addEventListener(
      'mousemove',
      this.handleMouseMove
    );
    
    // 4. 监听点击
    this.iframe.contentWindow?.addEventListener(
      'click',
      this.handleClick
    );
  }
  
  disable(): void {
    this.overlay.remove();
    this.removeInspectScript();
  }
  
  private handleMouseMove = (e: MouseEvent): void => {
    // 获取鼠标下的元素
    const element = this.getElementAtPoint(e.clientX, e.clientY);
    
    if (element && element !== this.highlightedElement) {
      // 更新高亮
      this.highlightElement(element);
    }
  };
  
  private handleClick = (e: MouseEvent): void => {
    e.preventDefault();
    e.stopPropagation();
    
    // 获取点击的元素
    const element = this.getElementAtPoint(e.clientX, e.clientY);
    
    if (element) {
      this.selectElement(element);
      this.showActionMenu(element, e.clientX, e.clientY);
    }
  };
  
  private getElementAtPoint(x: number, y: number): Element | null {
    // 通过postMessage获取元素
    return this.iframe.contentWindow?.document.elementFromPoint(x, y);
  }
  
  private highlightElement(element: Element): void {
    const rect = element.getBoundingClientRect();
    
    // 移动高亮框
    this.updateHighlight({
      left: rect.left,
      top: rect.top,
      width: rect.width,
      height: rect.height,
      tagName: element.tagName.toLowerCase(),
      classes: Array.from(element.classList),
    });
  }
  
  private selectElement(element: Element): void {
    this.selectedElement = element;
    
    // 计算选择器
    const selector = this.generateSelector(element);
    
    // 通知父窗口
    this.postToParent({
      type: 'inspect:select',
      selector,
      tagName: element.tagName.toLowerCase(),
      rect: element.getBoundingClientRect(),
      innerHTML: element.innerHTML.substring(0, 200),
    });
  }
  
  private generateSelector(element: Element): string {
    // 生成最优选择器
    if (element.id) {
      return `#${element.id}`;
    }
    
    if (element.className && typeof element.className === 'string') {
      const classes = element.className.trim().split(/\s+/);
      if (classes.length > 0) {
        return `${element.tagName.toLowerCase()}.${classes[0]}`;
      }
    }
    
    // 使用nth-child
    const parent = element.parentElement;
    if (parent) {
      const siblings = Array.from(parent.children);
      const index = siblings.indexOf(element) + 1;
      return `${element.tagName.toLowerCase()}:nth-child(${index})`;
    }
    
    return element.tagName.toLowerCase();
  }
}
```

```typescript
interface ConsoleMessage {
  type: 'log' | 'warn' | 'error' | 'info' | 'debug';
  args: unknown[];
  timestamp: number;
  stack?: string;
}

class ConsoleService {
  private messages: ConsoleMessage[] = [];
  private listeners: Set<(msg: ConsoleMessage) => void> = new Set();
  
  // 拦截iframe的console
  setupIframeConsole(iframe: HTMLIFrameElement): void {
    const script = `
      (function() {
        const originalConsole = {
          log: console.log.bind(console),
          warn: console.warn.bind(console),
          error: console.error.bind(console),
          info: console.info.bind(console),
          debug: console.debug.bind(console),
        };
        
        function proxyConsole(type, ...args) {
          originalConsole[type](...args);
          window.parent.postMessage({
            type: 'console',
            level: type,
            args: args.map(arg => {
              try {
                return JSON.stringify(arg, null, 2);
              } catch {
                return String(arg);
              }
            }),
            timestamp: Date.now(),
          }, '*');
        }
        
        console.log = (...args) => proxyConsole('log', ...args);
        console.warn = (...args) => proxyConsole('warn', ...args);
        console.error = (...args) => proxyConsole('error', ...args);
        console.info = (...args) => proxyConsole('info', ...args);
        console.debug = (...args) => proxyConsole('debug', ...args);
        
        window.onerror = (msg, src, line, col, error) => {
          window.parent.postMessage({
            type: 'console',
            level: 'error',
            args: [msg],
            stack: error?.stack,
            timestamp: Date.now(),
          }, '*');
        };
        
        window.onunhandledrejection = (event) => {
          window.parent.postMessage({
            type: 'console',
            level: 'error',
            args: ['Unhandled Promise Rejection:', event.reason],
            timestamp: Date.now(),
          }, '*');
        };
      })();
    `;
    
    iframe.contentDocument?.head.insertAdjacentHTML(
      'beforeend',
      `<script>${script}</script>`
    );
  }
  
  // 接收console消息
  handleMessage(event: MessageEvent): void {
    if (event.data?.type === 'console') {
      const message: ConsoleMessage = {
        type: event.data.level,
        args: event.data.args,
        timestamp: event.data.timestamp,
        stack: event.data.stack,
      };
      
      this.messages.push(message);
      this.emit(message);
    }
  }
  
  subscribe(listener: (msg: ConsoleMessage) => void): () => void {
    this.listeners.add(listener);
    return () => this.listeners.delete(listener);
  }
  
  private emit(message: ConsoleMessage): void {
    this.listeners.forEach(listener => listener(message));
  }
  
  getMessages(filter?: {
    type?: ConsoleMessage['type'];
    search?: string;
  }): ConsoleMessage[] {
    let filtered = this.messages;
    
    if (filter?.type) {
      filtered = filtered.filter(m => m.type === filter.type);
    }
    
    if (filter?.search) {
      const searchLower = filter.search.toLowerCase();
      filtered = filtered.filter(m => 
        m.args.some(arg => 
          typeof arg === 'string' && arg.toLowerCase().includes(searchLower)
        )
      );
    }
    
    return filtered;
  }
}
```

```typescript
class ViewModeManager {
  private state: ViewState;
  private listeners: Set<(state: ViewState) => void>;
  
  constructor() {
    this.state = this.loadState();
    this.listeners = new Set();
  }
  
  switchMode(mode: ViewMode): void {
    // 保存当前状态
    this.saveCurrentState();
    
    // 切换模式
    this.state.mode = mode;
    
    // 恢复目标模式状态
    this.restoreState(mode);
    
    // 通知更新
    this.notify();
    
    // 持久化
    this.persistState();
  }
  
  private saveCurrentState(): void {
    switch (this.state.mode) {
      case 'editor':
        this.state.editor = {
          activeFile: this.getActiveFile(),
          scrollPosition: this.getScrollPosition(),
          expandedFolders: this.getExpandedFolders(),
        };
        break;
      case 'preview':
        this.state.preview = {
          selectedElement: this.getSelectedElement(),
          devicePreset: this.getDevicePreset(),
          zoom: this.getZoom(),
        };
        break;
      case 'split':
        this.state.split.ratio = this.getSplitRatio();
        break;
    }
  }
  
  private restoreState(mode: ViewMode): void {
    switch (mode) {
      case 'editor':
        if (this.state.editor) {
          this.setActiveFile(this.state.editor.activeFile);
          this.setScrollPosition(this.state.editor.scrollPosition);
          this.setExpandedFolders(this.state.editor.expandedFolders);
        }
        break;
      case 'preview':
        if (this.state.preview) {
          this.setSelectedElement(this.state.preview.selectedElement);
          this.setDevicePreset(this.state.preview.devicePreset);
          this.setZoom(this.state.preview.zoom);
        }
        break;
      case 'split':
        if (this.state.split) {
          this.setSplitRatio(this.state.split.ratio);
        }
        break;
    }
  }
}
```

```tsx
  const containerRef = useRef<HTMLDivElement>(null);

    setIsDragging(true);

  };
  
  const handleMouseMove = useCallback((e: MouseEvent) => {
    if (!isDragging || !containerRef.current) return;
    
    const rect = containerRef.current.getBoundingClientRect();
    const newRatio = (e.clientX - rect.left) / rect.width;
    onRatioChange(Math.max(0.2, Math.min(0.8, newRatio)));
  }, [isDragging, onRatioChange]);
  
  const handleMouseUp = useCallback(() => {
    setIsDragging(false);
  }, []);
  
  useEffect(() => {
    if (isDragging) {
      document.addEventListener('mousemove', handleMouseMove);
      document.addEventListener('mouseup', handleMouseUp);
    }
    return () => {
      document.removeEventListener('mousemove', handleMouseMove);
      document.removeEventListener('mouseup', handleMouseUp);
    };
  }, [isDragging, handleMouseMove, handleMouseUp]);

    <div className="split-view" ref={containerRef}>

        className="split-editor"
```

#### 6.3 版本管理服务

```typescript
class VersionService {
  private storage: VersionStorage;
  private snapshotService: SnapshotService;
  
  // 创建新版本
  async createVersion(
    projectId: string,
    options: {
      name?: string;
      description?: string;
      trigger: 'manual' | 'auto' | 'deploy';
    }
  ): Promise<Version> {
    // 1. 生成快照
    const snapshot = await this.snapshotService.createSnapshot(projectId);
    
    // 2. 计算差异
    const previousVersion = await this.getLatestVersion(projectId);
    const stats = previousVersion 
      ? this.calculateDiff(snapshot, previousVersion.snapshot)
      : this.initialStats(snapshot);
    
    // 3. 创建版本
    const version: Version = {
      id: generateId(),
      projectId,
      name: options.name || this.generateVersionName(projectId),
      description: options.description || '',
      createdAt: Date.now(),
      createdBy: 'ai',
      trigger: options.trigger,
      snapshot,
      metadata: {
        parentVersion: previousVersion?.id,
        childVersions: [],
        branch: 'main',
        isHead: true,
      },
      stats,
    };
    
    // 4. 更新旧版本的isHead
    if (previousVersion) {
      await this.storage.update(previousVersion.id, {
        'metadata.childVersions': [...previousVersion.metadata.childVersions, version.id],
        'metadata.isHead': false,
      });
    }
    
    // 5. 保存新版本
    await this.storage.save(version);
    
    return version;
  }
  
  // 获取版本列表
  async listVersions(projectId: string): Promise<Version[]> {
    return this.storage.query({
      projectId,
      orderBy: 'createdAt',
      order: 'desc',
    });
  }
  
  // 切换版本
  async switchVersion(versionId: string): Promise<void> {
    const version = await this.storage.get(versionId);
    
    if (!version) {
      throw new Error('Version not found');
    }
    
    // 1. 保存当前工作状态
    await this.saveCurrentState();
    
    // 2. 恢复目标版本的快照
    await this.snapshotService.restoreSnapshot(version.snapshot);
    
    // 3. 通知UI刷新
    this.emit('version-switched', version);
  }
  
  // 对比两个版本
  async compareVersions(versionAId: string, versionBId: string): Promise<VersionDiff> {
    const [versionA, versionB] = await Promise.all([
      this.storage.get(versionAId),
      this.storage.get(versionBId),
    ]);
    
    if (!versionA || !versionB) {
      throw new Error('Version not found');
    }
    
    return this.computeDiff(versionA, versionB);
  }
  
  // 删除版本
  async deleteVersion(versionId: string): Promise<void> {
    const version = await this.storage.get(versionId);
    
    if (!version) {
      throw new Error('Version not found');
    }
    
    if (version.metadata.isHead) {
      throw new Error('Cannot delete head version');
    }
    
    // 删除快照文件
    await this.snapshotService.deleteSnapshot(version.snapshot);
    
    // 更新父版本的childVersions
    if (version.metadata.parentVersion) {
      const parent = await this.storage.get(version.metadata.parentVersion);
      if (parent) {
        await this.storage.update(parent.id, {
          'metadata.childVersions': parent.metadata.childVersions.filter(
            id => id !== versionId
          ),
        });
      }
    }
    
    // 删除版本记录
    await this.storage.delete(versionId);
  }
}
```

### 7. 实现决策考虑

#### 7.1 为什么选择iframe沙箱？

- **安全性**：完全隔离JavaScript执行环境
- **兼容性**：支持各种前端框架的渲染
- **简单性**：实现成本低，调试方便
- **灵活性**：可以独立管理iframe的生命周期

#### 7.2 热更新的技术选型

- **Vite HMR**：开发环境使用Vite原生HMR
- **WebSocket推送**：生产环境使用WebSocket通知更新
- **增量更新**：只更新变化的文件，减少带宽

#### 7.3 版本存储策略

- **差异存储**：只存储文件差异，而非完整快照
- **压缩存储**：使用gzip压缩存储内容
- **生命周期管理**：自动清理过旧版本

#### 7.4 Console收集的边界

- **API代理**：限制API请求的日志输出
- **敏感信息过滤**：自动过滤密码、token等敏感信息
- **性能考虑**：限制日志数量，防止内存溢出
