```typescript
// server/src/types/project.ts
export interface Project {
  id: string;
  name: string;
  userId: string;  // 用户标识
  sandboxId: string;
  files: FileInfo[];
  messages: Message[];
  createdAt: number;
  updatedAt: number;
}

// server/src/services/ProjectManager.ts
export class ProjectManager {
  private baseDir: string;
  private metaDir: string;
  
  constructor(baseDir: string) {
    this.baseDir = baseDir;
    this.metaDir = path.join(baseDir, '.meta');
    
    if (!fs.existsSync(this.metaDir)) {
      fs.mkdirSync(this.metaDir, { recursive: true });
    }
  }
  
  // 获取用户的项目列表
  listProjects(userId: string): Project[] {
    const userMetaDir = path.join(this.metaDir, userId);
    
    if (!fs.existsSync(userMetaDir)) {
      return [];
    }
    
    const projectFiles = fs.readdirSync(userMetaDir)
      .filter(f => f.endsWith('.json'));
    
    const projects: Project[] = projectFiles.map(file => {
      const filePath = path.join(userMetaDir, file);
      return JSON.parse(fs.readFileSync(filePath, 'utf8'));
    });
    
    // 按更新时间排序
    return projects.sort((a, b) => b.updatedAt - a.updatedAt);
  }
  
  // 保存项目元数据
  saveProject(project: Project): void {
    const userMetaDir = path.join(this.metaDir, project.userId);
    
    if (!fs.existsSync(userMetaDir)) {
      fs.mkdirSync(userMetaDir, { recursive: true });
    }
    
    const metaFile = path.join(userMetaDir, `${project.id}.json`);
    fs.writeFileSync(metaFile, JSON.stringify(project, null, 2));
    
    // 清理旧项目（最多5个）
    this.cleanupOldProjects(project.userId);
  }
  
  // 清理：超过5个时删除最旧的
  private cleanupOldProjects(userId: string): void {
    const projects = this.listProjects(userId);
    
    if (projects.length > 5) {
      const toDelete = projects.slice(5);
      
      toDelete.forEach(project => {
        // 删除元数据
        const userMetaDir = path.join(this.metaDir, userId);
        const metaFile = path.join(userMetaDir, `${project.id}.json`);
        if (fs.existsSync(metaFile)) {
          fs.unlinkSync(metaFile);
        }
        
        // 删除沙箱文件
        const sandboxDir = path.join(this.baseDir, project.sandboxId);
        if (fs.existsSync(sandboxDir)) {
          fs.rmSync(sandboxDir, { recursive: true, force: true });
        }
      });
    }
  }
  
  // 删除项目
  deleteProject(projectId: string, userId: string): void {
    const userMetaDir = path.join(this.metaDir, userId);
    const metaFile = path.join(userMetaDir, `${projectId}.json`);
    
    if (!fs.existsSync(metaFile)) {
      return;
    }
    
    const project: Project = JSON.parse(fs.readFileSync(metaFile, 'utf8'));
    
    // 删除元数据
    fs.unlinkSync(metaFile);
    
    // 删除沙箱文件
    const sandboxDir = path.join(this.baseDir, project.sandboxId);
    if (fs.existsSync(sandboxDir)) {
      fs.rmSync(sandboxDir, { recursive: true, force: true });
    }
  }
  
  // 获取单个项目
  getProject(projectId: string, userId: string): Project | null {
    const userMetaDir = path.join(this.metaDir, userId);
    const metaFile = path.join(userMetaDir, `${projectId}.json`);
    
    if (!fs.existsSync(metaFile)) {
      return null;
    }
    
    return JSON.parse(fs.readFileSync(metaFile, 'utf8'));
  }
}
```

```typescript
// client/src/utils/userId.ts
export function getUserId(): string {
  const STORAGE_KEY = 'atoms_user_id';
  
  let userId = localStorage.getItem(STORAGE_KEY);
  
  if (!userId) {
    // 生成新的用户 ID
    userId = crypto.randomUUID();
    localStorage.setItem(STORAGE_KEY, userId);
  }
  
  return userId;
}

// client/src/utils/projectRestore.ts
export function restoreProjectOnStartup() {
  const userId = getUserId();
  const lastProjectId = localStorage.getItem('atoms_last_project');
  
  if (lastProjectId) {
    // 尝试恢复最后一个项目
    socket?.emit('project:get', { projectId: lastProjectId, userId });
  }
  
  // 获取项目列表
  socket?.emit('project:list', { userId });
}
```

## 🧠 模块二：思维链与用户确认

### 2.1 设计理念

**核心流程**：
```
用户输入 → 理解分析 → 思维链展示 → 任务拆分 → 用户确认/修改 → 开始生成
```

**优点**：
- ✅ 前置确认意图，减少重做
- ✅ 透明化AI思考过程，建立信任
- ✅ 用户可调整方案，更符合需求
- ✅ 节省 Token（避免错误生成）

### 2.2 实现架构

```typescript

type ThinkingStepType =
```

```typescript
// server/src/websocket/SocketHandler.ts
class SocketHandler {
  private async handleThinkingChain(socket: Socket, content: string) {
    // 第一步：理解和规划
    socket.emit('thinking:step', {
      type: 'understanding',
      title: '正在理解您的需求...',
      content: '分析中...',
    });
    
    // 调用 LLM 进行思维链分析
    const thinkingResult = await llmService.thinkingChain(content);
    
    // 第二步：展示理解
    socket.emit('thinking:step', {
      type: 'understanding',
      title: '需求理解',
      content: thinkingResult.understanding,
      isCompleted: true,
    });
    
    // 第三步：方案规划
    socket.emit('thinking:step', {
      type: 'planning',
      title: '实现方案',
      content: thinkingResult.approach,
      isCompleted: true,
    });
    
    // 第四步：任务拆分
    socket.emit('thinking:step', {
      type: 'task_breakdown',
      title: '任务拆分',
      content: '共 ' + thinkingResult.tasks.length + ' 个任务',
      isCompleted: true,
    });
    
    // 发送任务列表
    socket.emit('thinking:tasks', {
      tasks: thinkingResult.tasks,
    });
    
    // 第五步：等待确认
    socket.emit('thinking:step', {
      type: 'confirming',
      title: '等待您的确认',
      content: '请确认方案，或提出修改意见',
    });
  }
  
  private async handleConfirm(socket: Socket) {
    // 用户确认，开始执行任务
    socket.emit('thinking:confirmed');
    await this.executeTasks(socket);
  }
}
```
