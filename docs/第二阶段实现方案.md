# 第二阶段实现方案

## 📋 核心需求概述

基于第一阶段完成，第二阶段聚焦：
1. **任务拆分模式** - AI 解析需求后展示任务列表，用户确认再执行
2. **用户意图确认** - 前置确认用户真实意图，避免重做
3. **记忆体系** - 两层记忆（用户级 + 项目级），Markdown格式持久化

---

## 🏗️ 模块一：任务拆分模式 + 用户意图确认

### 1.1 核心流程

```
用户输入需求
    ↓
AI 理解分析 → 生成任务列表
    ↓
展示任务列表 + 用户意图确认
    ↓
用户确认 / 修改意图 / 添加任务
    ↓
开始执行任务
    ↓
完成
```

```typescript
// 任务类型
type TaskType = 
  | 'create_file'    // 创建文件
  | 'update_file'    // 更新文件
  | 'delete_file'    // 删除文件
  | 'create_folder'  // 创建文件夹
  | 'run_command'    // 执行命令
  | 'preview'        // 预览确认
  | 'modify_style'   // 样式调整
  | 'add_interaction'; // 添加交互

// 单个任务
interface Task {
  id: string;
  type: TaskType;
  description: string;      // 任务描述
  files: string[];          // 相关文件列表
  content?: string;         // 文件内容（如果需要）
  estimatedTokens: number;  // 预估Token消耗
  dependencies: string[];   // 依赖任务ID
  status: 'pending' | 'in_progress' | 'completed' | 'failed';
}

// 用户意图分析
interface UserIntent {
  originalRequest: string;    // 原始需求
  understoodGoal: string;      // AI理解的目标
  scope: 'small' | 'medium' | 'large';  // 需求规模
  complexity: 'simple' | 'moderate' | 'complex';
  techStack: string[];         // 技术栈选择
  keyFeatures: string[];        // 核心功能点
  potentialIssues?: string[];  // 潜在问题
}

// 任务拆分结果
interface TaskBreakdown {
  id: string;
  userIntent: UserIntent;
  tasks: Task[];
  totalEstimatedTokens: number;
  createdAt: number;
  status: 'pending' | 'confirmed' | 'modified' | 'executing' | 'completed';
}
```

```typescript
// server/src/services/TaskAnalyzer.ts
export class TaskAnalyzer {
  private llmService: LLMService;
  
  constructor(llmService: LLMService) {
    this.llmService = llmService;
  }
  
  async analyzeRequest(userInput: string): Promise<TaskBreakdown> {
    // 调用 LLM 分析
    const response = await this.llmService.complete(
      TASK_BREAKDOWN_PROMPT + '\n\n用户需求：' + userInput
    );
    
    // 解析 JSON
    const result = JSON.parse(response);
    
    return {
      id: crypto.randomUUID(),
      userIntent: result.userIntent,
      tasks: result.tasks,
      totalEstimatedTokens: result.totalEstimatedTokens,
      createdAt: Date.now(),
      status: 'pending',
    };
  }
  
  generateConfirmationMessage(taskBreakdown: TaskBreakdown): string {
    const tasksList = taskBreakdown.tasks.map((task, i) => 
      `${i + 1}. [${task.type}] ${task.description}`
    ).join('\n');
    
    return INTENT_CONFIRMATION_PROMPT
      .replace('{userInput}', taskBreakdown.userIntent.originalRequest)
      .replace('{understoodGoal}', taskBreakdown.userIntent.understoodGoal)
      .replace('{techStack}', taskBreakdown.userIntent.techStack.join(', '))
      .replace('{tasksList}', tasksList);
  }
}

// server/src/websocket/SocketHandler.ts 扩展
class SocketHandler {
  private taskAnalyzer: TaskAnalyzer;
  private memoryManager: MemoryManager;
  
  async handleMessage(socket: Socket, content: string) {
    // 首次消息：分析任务
    const taskBreakdown = await this.taskAnalyzer.analyzeRequest(content);
    
    // 保存到记忆
    await this.memoryManager.addToProjectMemory(
      socket.data.projectId,
      {
        type: 'request',
        content,
        timestamp: Date.now(),
      }
    );
    
    // 发送任务分析结果
    socket.emit('task:breakdown', { taskBreakdown });
    
    // 发送确认消息
    const confirmation = this.taskAnalyzer.generateConfirmationMessage(taskBreakdown);
    socket.emit('chat:chunk', { content: confirmation });
    
    // 等待用户确认...
  }
  
  async handleTaskConfirm(socket: Socket) {
    // 用户确认，开始执行
    const taskBreakdown = socket.data.taskBreakdown;
    taskBreakdown.status = 'confirmed';
    
    await this.executeTasks(socket, taskBreakdown.tasks);
  }
}
```

### 2.4 记忆管理器实现

```typescript
// server/src/services/MemoryManager.ts
import * as fs from 'fs';
import * as path from 'path';

interface MemoryConfig {
  baseDir: string;
  userMemoryFile: string;
  projectMemoryFile: string;
}

interface MemoryEntry {
  type: string;
  content: string;
  timestamp: number;
}

interface UserMemory {
  userId: string;
  preferences: {
    techStack: Map<string, { frequency: number; lastUsed: number }>;
    uiStyle: { theme: string; colors: string[] };
    codingStyle: Record<string, string>;
  };
  projectHistory: Array<{
    projectId: string;
    name: string;
    date: number;
    status: string;
  }>;
  solvedProblems: Array<{
    problem: string;
    solution: string;
    date: number;
  }>;
  context: string;
}

interface ProjectMemory {
  projectId: string;
  userId: string;
  name: string;
  goal: string;
  techStack: string[];
  completedFeatures: string[];
  inProgressFeatures: string[];
  plannedFeatures: string[];
  fileStructure: string;
  conversationHistory: Array<{
    timestamp: number;
    userRequest: string;
    aiUnderstanding: string;
    tasks: string[];
    result: string;
  }>;
  keyDecisions: Array<{
    date: number;
    decision: string;
    reason: string;
  }>;
  pendingItems: string[];
  context: string;
}

export class MemoryManager {
  private baseDir: string;
  private usersDir: string;
  private projectsDir: string;
  
  constructor(baseDir: string) {
    this.baseDir = baseDir;
    this.usersDir = path.join(baseDir, 'users');
    this.projectsDir = path.join(baseDir, 'projects');
    
    this.ensureDirectories();
  }
  
  private ensureDirectories(): void {
    [this.usersDir, this.projectsDir].forEach(dir => {
      if (!fs.existsSync(dir)) {
        fs.mkdirSync(dir, { recursive: true });
      }
    });
  }
  
  // ============ 用户级记忆 ============
  
  async getUserMemory(userId: string): Promise<UserMemory> {
    const filePath = path.join(this.usersDir, `${userId}.md`);
    
    if (!fs.existsSync(filePath)) {
      return this.createEmptyUserMemory(userId);
    }
    
    return this.parseUserMemory(filePath);
  }
  
  async updateUserMemory(userId: string, updates: Partial<UserMemory>): Promise<void> {
    const memory = await this.getUserMemory(userId);
    const updated = { ...memory, ...updates, updatedAt: Date.now() };
    
    await this.saveUserMemory(userId, updated);
  }
  
  async addToUserMemory(userId: string, entry: MemoryEntry): Promise<void> {
    const memory = await this.getUserMemory(userId);
    
    // 根据类型更新不同的字段
    if (entry.type === 'tech_preference') {
      this.updateTechPreference(memory, entry.content);
    } else if (entry.type === 'solved_problem') {
      memory.solvedProblems.push({
        problem: entry.content,
        solution: '', // 待补充
        date: entry.timestamp,
      });
    }
    
    memory.updatedAt = Date.now();
    await this.saveUserMemory(userId, memory);
  }
  
  async buildUserContext(userId: string): Promise<string> {
    const memory = await this.getUserMemory(userId);
    
    // 生成上下文摘要
    const techPreferences = this.getTopTechnologies(memory);
    const recentProjects = memory.projectHistory.slice(-3);
    
    return `
## 用户偏好上下文

### 技术栈偏好
${techPreferences}

### 最近的3个项目
${recentProjects.map(p => `- ${p.name} (${new Date(p.date).toLocaleDateString()})`).join('\n')}

### 已知偏好
${memory.context || '无特殊偏好'}
`.trim();
  }
  
  // ============ 项目级记忆 ============
  
  async getProjectMemory(projectId: string): Promise<ProjectMemory | null> {
    const filePath = path.join(this.projectsDir, `${projectId}.md`);
    
    if (!fs.existsSync(filePath)) {
      return null;
    }
    
    return this.parseProjectMemory(filePath);
  }
  
  async createProjectMemory(projectId: string, userId: string, name: string): Promise<ProjectMemory> {
    const memory: ProjectMemory = {
      projectId,
      userId,
      name,
      goal: '',
      techStack: [],
      completedFeatures: [],
      inProgressFeatures: [],
      plannedFeatures: [],
      fileStructure: '',
      conversationHistory: [],
      keyDecisions: [],
      pendingItems: [],
      context: '',
    };
    
    await this.saveProjectMemory(projectId, memory);
    return memory;
  }
  
  async updateProjectMemory(projectId: string, updates: Partial<ProjectMemory>): Promise<void> {
    let memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return;
    }
    
    memory = { ...memory, ...updates, updatedAt: Date.now() };
    await this.saveProjectMemory(memory);
  }
  
  async addConversationToProject(
    projectId: string,
    conversation: {
      userRequest: string;
      aiUnderstanding: string;
      tasks: string[];
      result: string;
    }
  ): Promise<void> {
    const memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return;
    }
    
    memory.conversationHistory.push({
      ...conversation,
      timestamp: Date.now(),
    });
    
    memory.updatedAt = Date.now();
    await this.saveProjectMemory(projectId, memory);
  }
  
  async addKeyDecision(
    projectId: string,
    decision: string,
    reason: string
  ): Promise<void> {
    const memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return;
    }
    
    memory.keyDecisions.push({
      date: Date.now(),
      decision,
      reason,
    });
    
    memory.updatedAt = Date.now();
    await this.saveProjectMemory(projectId, memory);
  }
  
  async buildProjectContext(projectId: string): Promise<string> {
    const memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return '';
    }
    
    return `
## 项目上下文

### 基本信息
- **项目名称**: ${memory.name}
- **目标**: ${memory.goal || '未定义'}
- **技术栈**: ${memory.techStack.join(', ') || '未选择'}

### 文件结构
${memory.fileStructure || '未创建'}

### 已完成功能
${memory.completedFeatures.map(f => `- ✅ ${f}`).join('\n')}

### 进行中功能
${memory.inProgressFeatures.map(f => `- 🔄 ${f}`).join('\n')}

### 计划中功能
${memory.plannedFeatures.map(f => `- 📋 ${f}`).join('\n')}

### 关键决策
${memory.keyDecisions.map(d => `- **${new Date(d.date).toLocaleDateString()}**: ${d.decision}`).join('\n')}

### 最近的对话
${memory.conversationHistory.slice(-2).map(c => `
**用户**: ${c.userRequest}
**AI理解**: ${c.aiUnderstanding}
**结果**: ${c.result}
`).join('\n---\n')}

### 上下文摘要
${memory.context || '无额外上下文'}
`.trim();
  }
  
  // ============ 上下文构建 ============
  
  async buildContext(userId: string, projectId?: string): Promise<string> {
    const contexts: string[] = [];
    
    // 用户级上下文
    const userContext = await this.buildUserContext(userId);
    if (userContext) {
      contexts.push(userContext);
    }
    
    // 项目级上下文
    if (projectId) {
      const projectContext = await this.buildProjectContext(projectId);
      if (projectContext) {
        contexts.push(projectContext);
      }
    }
    
    return contexts.join('\n\n---\n\n');
  }
  
  // ============ 私有方法 ============
  
  private createEmptyUserMemory(userId: string): UserMemory {
    return {
      userId,
      preferences: {
        techStack: new Map(),
        uiStyle: { theme: 'auto', colors: [] },
        codingStyle: {},
      },

  private extractField(content: string, field: string): string {
    const match = content.match(new RegExp(`\\*\\*${field}\\*\\*:\\s*(.+)`));
    return match ? match[1].trim() : '';
  }
  
  private extractSection(content: string, section: string): string {
    const match = content.match(new RegExp(`### ${section}\\s*\\n([\\s\\S]*?)(?=###|\\n---)`, 'i'));
    return match ? match[1].trim() : '';
  }
  
  private extractList(content: string, section: string): string[] {
    const sectionContent = this.extractSection(content, section);
    const items = sectionContent.match(/^- (.+)$/gm);
    return items ? items.map(item => item.replace(/^- /, '').trim()) : [];
  }
  
  private extractFeatures(content: string, type: string): string[] {
    const match = content.match(new RegExp(`- ✅ 已完成功能\\s*\\n([\\s\\S]*?)(?=- 🔄|- 📋|\\n###)`, 'i'));
    if (!match) return [];
    
    const items = match[1].match(/^- (.+)$/gm);
    return items ? items.map(item => item.replace(/^- /, '').trim()) : [];
  }
  
  private extractPendingItems(content: string): string[] {
    const match = content.match(/## 待处理事项\\s*\\n([\\s\\S]*?)(?=##|\\n---)/);
    if (!match) return [];
    
    const items = match[1].match(/^- \\[ \\] (.+)$/gm);
    return items ? items.map(item => item.replace(/^- \[ \] /, '').trim()) : [];
  }
  
  private extractCodeBlock(content: string): string {
    const match = content.match(/```\\n([\\s\\S]*?)```/);
    return match ? match[1].trim() : '';
  }
  
  private extractContext(content: string): string {
    const match = content.match(/<auto_generated_summary>([\\s\\S]*?)<\\/auto_generated_summary>/);
    return match ? match[1].trim() : '';
  }
  
  private parseTechStack(content: string): Map<string, { frequency: number; lastUsed: number }> {
    const techMap = new Map();
    const rows = content.match(/\\| [^|]+ \\| [^|]+ \\| [^|]+ \\|/g) || [];
    
    rows.forEach(row => {
      const cols = row.split('|').map(c => c.trim()).filter(c => c);
      if (cols.length >= 3) {
        const [tech, freq, date] = cols;
        const frequency = freq === '高' ? 3 : freq === '中' ? 2 : 1;
        techMap.set(tech, {
          frequency,
          lastUsed: new Date(date).getTime(),
        });
      }
    });
    
    return techMap;
  }
  
  private parseUiStyle(content: string): { theme: string; colors: string[] } {
    return {
      theme: this.extractField(content, '主题') || 'auto',
      colors: this.extractField(content, '主要颜色')?.split(',').map(c => c.trim()) || [],
    };
  }
  
  private parseCodingStyle(content: string): Record<string, string> {
    const style: Record<string, string> = {};
    const match = content.match(/## 编码习惯\\s*\\n([\\s\\S]*?)(?=##|\\n---)/i);
    
    if (match) {
      const lines = match[1].match(/^- (.+): (.+)$/gm);
      lines?.forEach(line => {
        const [, key, value] = line.match(/^- (.+): (.+)$/) || [];
        if (key && value) {
          style[key] = value;
        }
      });
    }
    
    return style;
  }

      .sort((a, b) => b[1].frequency - a[1].frequency);
```

## 📝 补充说明

### 为什么选择 Markdown 格式？

1. **人类可读** - 便于调试和手动修改
2. **LLM 友好** - 可以直接作为 context
3. **版本控制友好** - 如果需要 Git 管理
4. **简单可靠** - 不需要数据库，文件系统足够
5. **易于备份** - 可以直接复制文件

### 记忆更新策略

采用**增量更新 + 定期总结**的策略：
- 每次对话后增量更新相关字段
- 对话历史只保留最近 N 条（N 可配置）
- 定期生成新的上下文摘要（可选）

### 潜在问题

1. **并发写入** - 多用户同时修改需要加锁
2. **文件损坏** - 需要备份机制
3. **存储上限** - 需要定期清理旧数据
4. **隐私问题** - 记忆涉及用户数据，需要加密存储（可选）

这些问题可以在后续迭代中解决，Phase 1 先实现基本功能。
