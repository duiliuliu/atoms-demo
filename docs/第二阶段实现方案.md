## 1.4 项目管理器实现

```typescript
// server/src/services/ProjectManager.ts
import * as fs from 'fs';
import * as path from 'path';
import { getSandboxBaseDir } from '../config/env.js';

const MAX_PROJECTS = 5;

export class ProjectManager {
  private baseDir: string;
  private metaDir: string;
  
  constructor(baseDir?: string) {
    this.baseDir = baseDir || getSandboxBaseDir();
    this.metaDir = path.join(this.baseDir, '.meta');
    
    if (!fs.existsSync(this.metaDir)) {
      fs.mkdirSync(this.metaDir, { recursive: true });
    }
  }
  
  // 获取用户的项目列表
  listProjects(userId: string): Project[] {
    const userMetaDir = path.join(this.metaDir, userId);
    
    if (!fs.existsSync(userMetaDir)) {
      return [];
    }
    
    const projectFiles = fs.readdirSync(userMetaDir)
      .filter(f => f.endsWith('.json'));
    
    const projects: Project[] = projectFiles
      .map(file => {
        const filePath = path.join(userMetaDir, file);
        try {
          return JSON.parse(fs.readFileSync(filePath, 'utf-8')) as Project;
        } catch {
          return null;
        }
      })
      .filter((p): p is Project => p !== null);
    
    // 按最后访问时间排序
    return projects.sort((a, b) => (b.lastVisitedAt || 0) - (a.lastVisitedAt || 0));
  }
  
  // 获取单个项目
  getProject(projectId: string, userId: string): Project | null {
    const userMetaDir = path.join(this.metaDir, userId);
    const metaFile = path.join(userMetaDir, `${projectId}.json`);
    
    if (!fs.existsSync(metaFile)) {
      return null;
    }
    
    try {
      return JSON.parse(fs.readFileSync(metaFile, 'utf-8')) as Project;
    } catch {
      return null;
    }
  }
  
  // 创建新项目
  async createProject(userId: string, name: string): Promise<Project> {
    // 清理旧项目（如果超过5个）
    await this.cleanupOldProjects(userId);
    
    const projectId = crypto.randomUUID();
    
    const project: Project = {
      id: projectId,
      userId,
      name,
      sandboxId: projectId,
      messages: [],
      files: [],
      createdAt: Date.now(),
      updatedAt: Date.now(),
      lastVisitedAt: Date.now(),
      status: 'active',
    };
    
    await this.saveProject(project);
    
    return project;
  }
  
  // 保存项目
  async saveProject(project: Project): Promise<void> {
    const userMetaDir = path.join(this.metaDir, project.userId);
    
    if (!fs.existsSync(userMetaDir)) {
      fs.mkdirSync(userMetaDir, { recursive: true });
    }
    
    const metaFile = path.join(userMetaDir, `${project.id}.json`);
    project.updatedAt = Date.now();
    
    fs.writeFileSync(metaFile, JSON.stringify(project, null, 2), 'utf-8');
  }
  
  // 更新项目访问时间
  async touchProject(projectId: string, userId: string): Promise<void> {
    const project = this.getProject(projectId, userId);
    
    if (project) {
      project.lastVisitedAt = Date.now();
      await this.saveProject(project);
    }
  }
  
  // 更新项目名称
  async renameProject(projectId: string, userId: string, name: string): Promise<void> {
    const project = this.getProject(projectId, userId);
    
    if (project) {
      project.name = name;
      await this.saveProject(project);
    }
  }
  
  // 删除项目
  async deleteProject(projectId: string, userId: string): Promise<void> {
    const project = this.getProject(projectId, userId);
    
    if (!project) {
      return;
    }
    
    // 删除元数据
    const userMetaDir = path.join(this.metaDir, userId);
    const metaFile = path.join(userMetaDir, `${projectId}.json`);
    
    if (fs.existsSync(metaFile)) {
      fs.unlinkSync(metaFile);
    }
    
    // 删除沙箱文件
    const sandboxDir = path.join(this.baseDir, project.sandboxId);
    if (fs.existsSync(sandboxDir)) {
      fs.rmSync(sandboxDir, { recursive: true, force: true });
    }
  }
  
  // 清理：超过5个时删除最旧的
  private async cleanupOldProjects(userId: string): Promise<void> {
    const projects = this.listProjects(userId);
    
    if (projects.length >= MAX_PROJECTS) {
      const toDelete = projects.slice(MAX_PROJECTS - 1);
      
      for (const project of toDelete) {
        await this.deleteProject(project.id, userId);
        console.log(`[ProjectManager] Cleaned up old project: ${project.name}`);
      }
    }
  }
  
  // 更新项目消息
  async addMessage(projectId: string, userId: string, message: Message): Promise<void> {
    const project = this.getProject(projectId, userId);
    
    if (project) {
      project.messages.push(message);
      await this.saveProject(project);
    }
  }
  
  // 更新项目文件
  async updateFiles(projectId: string, userId: string, files: FileInfo[]): Promise<void> {
    const project = this.getProject(projectId, userId);
    
    if (project) {
      project.files = files;
      await this.saveProject(project);
    }
  }
}
```

## 2.4 任务分析器实现

```typescript
// server/src/services/TaskAnalyzer.ts
import { LLMService } from './llm/LLMService.js';

export class TaskAnalyzer {
  private llmService: LLMService;
  
  constructor(llmService: LLMService) {
    this.llmService = llmService;
  }
  
  async analyzeRequest(userInput: string): Promise<TaskBreakdown> {
    const prompt = `${TASK_BREAKDOWN_PROMPT}\n\n用户需求：${userInput}`;
    
    const response = await this.llmService.complete(prompt);
    
    let result: any;
    try {
      // 尝试解析 JSON
      const jsonMatch = response.match(/\{[\s\S]*\}/);
      if (jsonMatch) {
        result = JSON.parse(jsonMatch[0]);
      } else {
        throw new Error('No JSON found in response');
      }
    } catch (error) {
      // 如果解析失败，使用默认结构
      console.error('[TaskAnalyzer] Failed to parse LLM response:', error);
      result = {
        userIntent: {
          understoodGoal: userInput,
          scope: 'small',
          complexity: 'simple',
          techStack: ['HTML', 'CSS', 'JavaScript'],
          keyFeatures: [],
        },
        tasks: [{
          id: '1',
          type: 'create_file',
          description: '创建基础文件',
          files: ['index.html'],
          estimatedTokens: 500,
          dependencies: [],
        }],
        totalEstimatedTokens: 500,
      };
    }
    
    return {
      id: crypto.randomUUID(),
      userIntent: {
        originalRequest: userInput,
        ...result.userIntent,
      },
      tasks: result.tasks || [],
      totalEstimatedTokens: result.totalEstimatedTokens || 0,
      createdAt: Date.now(),
      status: 'pending',
    };
  }
  
  generateConfirmationMessage(taskBreakdown: TaskBreakdown): string {
    const tasksList = taskBreakdown.tasks.map((task, i) => 
      `${i + 1}. [${task.type}] ${task.description}`
    ).join('\n');
    
    const potentialIssues = taskBreakdown.userIntent.potentialIssues?.length
      ? `\n\n⚠️ **需要注意**：\n${taskBreakdown.userIntent.potentialIssues.map(p => `- ${p}`).join('\n')}`
      : '';
    
    return INTENT_CONFIRMATION_TEMPLATE
      .replace('{understoodGoal}', taskBreakdown.userIntent.understoodGoal)
      .replace('{scope}', taskBreakdown.userIntent.scope)
      .replace('{complexity}', taskBreakdown.userIntent.complexity)
      .replace('{techStack}', taskBreakdown.userIntent.techStack.join(', '))
      .replace('{keyFeatures}', taskBreakdown.userIntent.keyFeatures.join('\n'))
      .replace('{potentialIssues}', potentialIssues)
      .replace('{tasksList}', tasksList);
  }
}
```

# 📦 模块三：记忆体系设计

## 3.1 设计理念

**两层记忆结构**：
1. **项目级记忆** - 每个项目的上下文、技术栈、已完成的任务
2. **用户级记忆** - 跨项目的用户偏好、常用技术栈、编码风格

**格式选择**：Markdown
- 人类可读
- 便于调试
- 可版本控制
- LLM 友好（可直接作为 context）

## 3.4 记忆管理器实现

```typescript
// server/src/services/MemoryManager.ts
import * as fs from 'fs';
import * as path from 'path';
import { getSandboxBaseDir } from '../config/env.js';

interface MemoryConfig {
  baseDir: string;
  usersDir: string;
  projectsDir: string;
}

interface UserMemory {
  userId: string;
  createdAt: number;
  updatedAt: number;
  preferences: {
    techStack: Array<{ tech: string; frequency: number; lastUsed: number }>;
    uiStyle: { theme: string; colors: string[] };
    codingStyle: Record<string, string>;
  };
  projectHistory: Array<{
    projectId: string;
    name: string;
    date: number;
    status: string;
  }>;
  solvedProblems: Array<{
    problem: string;
    solution: string;
    date: number;
  }>;
  context: string;
}

interface ProjectMemory {
  projectId: string;
  userId: string;
  name: string;
  createdAt: number;
  updatedAt: number;
  goal: string;
  techStack: string[];
  completedFeatures: string[];
  inProgressFeatures: string[];
  plannedFeatures: string[];
  fileStructure: string;
  conversationHistory: Array<{
    timestamp: number;
    userRequest: string;
    aiUnderstanding: string;
    tasks: string[];
    result: string;
  }>;
  keyDecisions: Array<{
    date: number;
    decision: string;
    reason: string;
  }>;
  pendingItems: string[];
  context: string;
}

export class MemoryManager {
  private baseDir: string;
  private usersDir: string;
  private projectsDir: string;
  
  constructor(baseDir?: string) {
    // 记忆目录在沙箱目录的上一级
    this.baseDir = baseDir || path.join(getSandboxBaseDir(), '..', 'atoms-memory');
    this.usersDir = path.join(this.baseDir, 'users');
    this.projectsDir = path.join(this.baseDir, 'projects');
    
    this.ensureDirectories();
  }
  
  private ensureDirectories(): void {
    [this.usersDir, this.projectsDir].forEach(dir => {
      if (!fs.existsSync(dir)) {
        fs.mkdirSync(dir, { recursive: true });
      }
    });
  }
  
  // ============ 用户级记忆 ============
  
  async getUserMemory(userId: string): Promise<UserMemory> {
    const filePath = path.join(this.usersDir, `${userId}.md`);
    
    if (!fs.existsSync(filePath)) {
      return this.createEmptyUserMemory(userId);
    }
    
    return this.parseUserMemory(filePath);
  }
  
  async saveUserMemory(userId: string, memory: UserMemory): Promise<void> {
    const filePath = path.join(this.usersDir, `${userId}.md`);
    const content = this.serializeUserMemory(memory);
    
    fs.writeFileSync(filePath, content, 'utf-8');
  }
  
  async updateUserMemory(userId: string, updates: Partial<UserMemory>): Promise<void> {
    const memory = await this.getUserMemory(userId);
    const updated = { 
      ...memory, 
      ...updates, 
      userId, // 确保 userId 不被覆盖
      updatedAt: Date.now() 
    };
    
    await this.saveUserMemory(userId, updated);
  }
  
  async addTechPreference(userId: string, tech: string): Promise<void> {
    const memory = await this.getUserMemory(userId);
    
    const existing = memory.preferences.techStack.find(t => t.tech === tech);
    if (existing) {
      existing.frequency += 1;
      existing.lastUsed = Date.now();
    } else {
      memory.preferences.techStack.push({
        tech,
        frequency: 1,
        lastUsed: Date.now(),
      });
    }
    
    memory.updatedAt = Date.now();
    await this.saveUserMemory(userId, memory);
  }
  
  async addProjectToHistory(userId: string, projectId: string, name: string): Promise<void> {
    const memory = await this.getUserMemory(userId);
    
    memory.projectHistory.push({
      projectId,
      name,
      date: Date.now(),
      status: 'active',
    });
    
    // 只保留最近 10 个项目
    if (memory.projectHistory.length > 10) {
      memory.projectHistory = memory.projectHistory.slice(-10);
    }
    
    memory.updatedAt = Date.now();
    await this.saveUserMemory(userId, memory);
  }
  
  async buildUserContext(userId: string): Promise<string> {
    const memory = await this.getUserMemory(userId);
    
    if (!memory || memory.projectHistory.length === 0) {
      return '';
    }
    
    const topTechs = memory.preferences.techStack
      .sort((a, b) => b.frequency - a.frequency)
      .slice(0, 5);
    
    const recentProjects = memory.projectHistory.slice(-3);
    
    return `
## 用户偏好上下文

### 技术栈偏好
${topTechs.map(t => `- ${t.tech} (${t.frequency > 2 ? '常用' : '偶尔使用'})`).join('\n')}

### 最近的3个项目
${recentProjects.map(p => `- ${p.name} (${new Date(p.date).toLocaleDateString()})`).join('\n')}

### 已知偏好
${memory.context || '无特殊偏好'}
`.trim();
  }
  
  // ============ 项目级记忆 ============
  
  async getProjectMemory(projectId: string): Promise<ProjectMemory | null> {
    const filePath = path.join(this.projectsDir, `${projectId}.md`);
    
    if (!fs.existsSync(filePath)) {
      return null;
    }
    
    return this.parseProjectMemory(filePath);
  }
  
  async createProjectMemory(projectId: string, userId: string, name: string): Promise<ProjectMemory> {
    const memory: ProjectMemory = {
      projectId,
      userId,
      name,
      createdAt: Date.now(),
      updatedAt: Date.now(),
      goal: '',
      techStack: [],
      completedFeatures: [],
      inProgressFeatures: [],
      plannedFeatures: [],
      fileStructure: '',
      conversationHistory: [],
      keyDecisions: [],
      pendingItems: [],
      context: '',
    };
    
    await this.saveProjectMemory(projectId, memory);
    return memory;
  }
  
  async saveProjectMemory(projectId: string, memory: ProjectMemory): Promise<void> {
    const filePath = path.join(this.projectsDir, `${projectId}.md`);
    const content = this.serializeProjectMemory(memory);
    
    fs.writeFileSync(filePath, content, 'utf-8');
  }
  
  async updateProjectMemory(projectId: string, updates: Partial<ProjectMemory>): Promise<void> {
    let memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return;
    }
    
    memory = { ...memory, ...updates, updatedAt: Date.now() };
    await this.saveProjectMemory(projectId, memory);
  }
  
  async addConversation(projectId: string, conversation: {
    userRequest: string;
    aiUnderstanding: string;
    tasks: string[];
    result: string;
  }): Promise<void> {
    const memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return;
    }
    
    memory.conversationHistory.push({
      ...conversation,
      timestamp: Date.now(),
    });
    
    // 只保留最近 20 条对话
    if (memory.conversationHistory.length > 20) {
      memory.conversationHistory = memory.conversationHistory.slice(-20);
    }
    
    memory.updatedAt = Date.now();
    await this.saveProjectMemory(projectId, memory);
  }
  
  async addCompletedFeature(projectId: string, feature: string): Promise<void> {
    const memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return;
    }
    
    // 从进行中移到已完成
    const inProgressIndex = memory.inProgressFeatures.indexOf(feature);
    if (inProgressIndex > -1) {
      memory.inProgressFeatures.splice(inProgressIndex, 1);
    }
    
    if (!memory.completedFeatures.includes(feature)) {
      memory.completedFeatures.push(feature);
    }
    
    memory.updatedAt = Date.now();
    await this.saveProjectMemory(projectId, memory);
  }
  
  async addPendingItem(projectId: string, item: string): Promise<void> {
    const memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return;
    }
    
    if (!memory.pendingItems.includes(item)) {
      memory.pendingItems.push(item);
    }
    
    memory.updatedAt = Date.now();
    await this.saveProjectMemory(projectId, memory);
  }
  
  async buildProjectContext(projectId: string): Promise<string> {
    const memory = await this.getProjectMemory(projectId);
    
    if (!memory) {
      return '';
    }
    
    return `
## 项目上下文

### 基本信息
- **项目名称**: ${memory.name}
- **目标**: ${memory.goal || '未定义'}
- **技术栈**: ${memory.techStack.join(', ') || '未选择'}

### 文件结构
\`\`\`
${memory.fileStructure || '未创建'}
\`\`\`

### 已完成功能
${memory.completedFeatures.map(f => `- ✅ ${f}`).join('\n')}

### 进行中功能
${memory.inProgressFeatures.map(f => `- 🔄 ${f}`).join('\n')}

### 计划中功能
${memory.plannedFeatures.map(f => `- 📋 ${f}`).join('\n')}

### 关键决策
${memory.keyDecisions.map(d => 
  `- **${new Date(d.date).toLocaleDateString()}**: ${d.decision}`
).join('\n')}

### 最近的对话
${memory.conversationHistory.slice(-2).map(c => `
**用户**: ${c.userRequest}
**AI理解**: ${c.aiUnderstanding}
**结果**: ${c.result}
`).join('\n---\n')}

### 上下文摘要
${memory.context || '无额外上下文'}
`.trim();
  }
  
  // ============ 上下文构建 ============
  
  async buildContext(userId: string, projectId?: string): Promise<string> {
    const contexts: string[] = [];
    
    // 用户级上下文
    const userContext = await this.buildUserContext(userId);
    if (userContext) {
      contexts.push(userContext);
    }
    
    // 项目级上下文
    if (projectId) {
      const projectContext = await this.buildProjectContext(projectId);
      if (projectContext) {
        contexts.push(projectContext);
      }
    }
    
    return contexts.join('\n\n---\n\n');
  }
  
  // ============ 序列化/反序列化 ============
  
  private createEmptyUserMemory(userId: string): UserMemory {
    return {
      userId,
      createdAt: Date.now(),
      updatedAt: Date.now(),
      preferences: {
        techStack: [],
        uiStyle: { theme: 'auto', colors: [] },
        codingStyle: {},
      },
      projectHistory: [],
      solvedProblems: [],
      context: '',
    };
  }
  
  private parseUserMemory(filePath: string): UserMemory {
    const content = fs.readFileSync(filePath, 'utf-8');
    const contextMatch = content.match(/<auto_generated_summary>([\s\S]*?)<\/auto_generated_summary>/);
    
    return {
      userId: this.extractField(content, 'userId'),
      createdAt: new Date(this.extractField(content, '创建时间') || Date.now()).getTime(),
      updatedAt: Date.now(),
      preferences: {
        techStack: this.parseTechStack(content),
        uiStyle: this.parseUiStyle(content),
        codingStyle: this.parseCodingStyle(content),
      },
      projectHistory: this.parseProjectHistory(content),
      solvedProblems: this.parseSolvedProblems(content),
      context: contextMatch ? contextMatch[1].trim() : '',
    };
  }
  
  private serializeUserMemory(memory: UserMemory): string {
    const techStackTable = memory.preferences.techStack
      .sort((a, b) => b.frequency - a.frequency)
      .map(t => {
        const freq = t.frequency > 2 ? '高' : t.frequency > 1 ? '中' : '低';
        return `| ${t.tech} | ${freq} | ${new Date(t.lastUsed).toLocaleDateString()} |`;
      }).join('\n');
    
    return `---
type: user_memory
version: 1.0
userId: ${memory.userId}
createdAt: ${new Date(memory.createdAt).toISOString()}
updatedAt: ${new Date(memory.updatedAt).toISOString()}
---

# 用户记忆

## 基本信息
- **用户ID**: ${memory.userId}
- **创建时间**: ${new Date(memory.createdAt).toLocaleString()}
- **最后活跃**: ${new Date(memory.updatedAt).toLocaleString()}

## 偏好设置

### 技术栈偏好
| 技术 | 使用频率 | 最后使用时间 |
|------|----------|--------------|
${techStackTable || '| - | - | - |'}

### UI 风格偏好
- **主题**: ${memory.preferences.uiStyle.theme}
- **主要颜色**: ${memory.preferences.uiStyle.colors.join(', ') || '未设置'}

### 编码习惯
${Object.entries(memory.preferences.codingStyle).map(([k, v]) => `- ${k}: ${v}`).join('\n') || '未设置'}

## 项目历史

### 最近项目
${memory.projectHistory.slice(-5).reverse().map(p => 
`1. **${p.name}** - ${new Date(p.date).toLocaleDateString()}
   - 完成状态: ${p.status}`
).join('\n\n') || '暂无项目'}

## 学习记录

### 已解决的问题
${memory.solvedProblems.slice(-5).map(p => 
`- ${p.problem} → ${p.solution} (${new Date(p.date).toLocaleDateString()})`
).join('\n') || '暂无记录'}

## 上下文摘要

<auto_generated_summary>
${memory.context || '新用户，暂无偏好记录。'}
</auto_generated_summary>

---

*此文档最后更新于 ${new Date(memory.updatedAt).toLocaleString()}*
`;
  }
  
  private parseProjectMemory(filePath: string): ProjectMemory {
    const content = fs.readFileSync(filePath, 'utf-8');
    
    return {
      projectId: this.extractField(content, 'projectId'),
      userId: this.extractField(content, 'userId'),
      name: this.extractField(content, '项目名称') || '',
      createdAt: new Date(this.extractField(content, '创建时间') || Date.now()).getTime(),

      context: this.extractContext(content),

  }

  // ============ 辅助方法 ============
  
  private extractField(content: string, field: string): string {
    const match = content.match(new RegExp(`\\*\\*${field}\\*\\*:\\s*(.+)`));
    return match ? match[1].trim() : '';
  }
  
  private extractSection(content: string, section: string): string {
    const match = content.match(new RegExp(`### ${section}\\s*\\n([\\s\\S]*?)(?=###|\\n---)`, 'i'));
    return match ? match[1].trim() : '';
  }
  
  private extractList(content: string, section: string): string[] {
    const sectionContent = this.extractSection(content, section);
    const items = sectionContent.match(/^- (.+)$/gm);
    return items ? items.map(item => item.replace(/^- /, '').trim()) : [];
  }
  
  private extractFeatures(content: string, type: string): string[] {
    const patterns: Record<string, RegExp> = {
      '已完成功能': /- ✅ 已完成功能\s*\n([\s\S]*?)(?=- 🔄|- 📋|\n###)/i,
      '进行中功能': /- 🔄 进行中\s*\n([\s\S]*?)(?=- ✅|- 📋|\n###)/i,
      '计划中功能': /- 📋 计划中\s*\n([\s\S]*?)(?=##)/i,
    };
    
    const match = content.match(patterns[type]);
    if (!match) return [];
    
    const items = match[1].match(/^- (.+)$/gm);
    return items ? items.map(item => item.replace(/^- /, '').trim()) : [];
  }
  
  private extractPendingItems(content: string): string[] {
    const match = content.match(/## 待处理事项\s*\n([\s\S]*?)(?=##|\n---)/);
    if (!match) return [];
    
    const items = match[1].match(/^- \[ \] (.+)$/gm);
    return items ? items.map(item => item.replace(/^- \[ \] /, '').trim()) : [];
  }
  
  private extractCodeBlock(content: string): string {
    const match = content.match(/```\n([\s\S]*?)```/);
    return match ? match[1].trim() : '';
  }
  
  private extractContext(content: string): string {
    const match = content.match(/<auto_generated_summary>([\s\S]*?)<\/auto_generated_summary>/);
    return match ? match[1].trim() : '';
  }
  
  private parseTechStack(content: string): Array<{ tech: string; frequency: number; lastUsed: number }> {
    const techs: Array<{ tech: string; frequency: number; lastUsed: number }> = [];
    const rows = content.match(/\| [^|]+ \| [^|]+ \| [^|]+ \|/g) || [];
    
    rows.forEach(row => {
      const cols = row.split('|').map(c => c.trim()).filter(c => c);
      if (cols.length >= 3) {
        const [tech, freq, date] = cols;
        const frequency = freq === '高' ? 3 : freq === '中' ? 2 : 1;
        techs.push({
          tech,
          frequency,
          lastUsed: new Date(date).getTime(),
        });
      }
    });
    
    return techs;
  }
  
  private parseUiStyle(content: string): { theme: string; colors: string[] } {
    return {
      theme: this.extractField(content, '主题') || 'auto',
      colors: this.extractField(content, '主要颜色')?.split(',').map(c => c.trim()) || [],
    };
  }
  
  private parseCodingStyle(content: string): Record<string, string> {
    const style: Record<string, string> = {};
    const match = content.match(/## 编码习惯\s*\n([\s\S]*?)(?=##|\n---)/i);
    
    if (match) {
      const lines = match[1].match(/^- (.+): (.+)$/gm);
      lines?.forEach(line => {
        const parts = line.replace(/^- /, '').split(':');
        if (parts.length === 2) {
          style[parts[0].trim()] = parts[1].trim();
        }
      });
    }
    
    return style;
  }
  
  private parseProjectHistory(content: string): Array<{ projectId: string; name: string; date: number; status: string }> {
    return [];
  }
  
  private parseSolvedProblems(content: string): Array<{ problem: string; solution: string; date: number }> {
    return [];
  }
  
  private parseConversationHistory(content: string): Array<any> {
    return [];
  }
  
  private parseKeyDecisions(content: string): Array<{ date: number; decision: string; reason: string }> {
    return [];
  }
}
```

## 📝 补充说明

### 为什么选择 Markdown 格式？

1. **人类可读** - 便于调试和手动修改
2. **LLM 友好** - 可以直接作为 context
3. **版本控制友好** - 如果需要 Git 管理
4. **简单可靠** - 不需要数据库，文件系统足够
5. **易于备份** - 可以直接复制文件

### 目录结构设计

- **生产环境**：`/tmp/atoms-memory/`
- **开发环境**：`project-root/atoms-memory/`

与沙箱目录平行，便于管理：
```
/tmp/
├── atoms-sandbox/     # 沙箱文件
└── atoms-memory/      # 记忆文件
```

### 潜在问题

1. **并发写入** - 多用户同时修改需要加锁（Phase 2 再处理）
2. **文件损坏** - 需要备份机制（Phase 2 再处理）
3. **存储上限** - 需要定期清理旧数据（Phase 2 再处理）

Phase 1 先实现基本功能，上线验证后再迭代。
